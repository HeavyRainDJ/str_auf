```python
import sys
import os
from PyPDF2 import PdfReader

# 1. Разработка регулярных выражений (Task 1)

def get_regex_pattern():
    """
    Возвращает скомпилированное регулярное выражение для поиска
    сочетаний слов и чисел, разделенных ОБЫЧНЫМ пробелом (или табуляцией),
    а не неразрывным.
    
    Ключевые слова: рисунок, рис, таблица, табл, от, до, менее, свыше, более.
    """
    # (?i) - флаг игнорирования регистра
    # \b - граница слова (чтобы не искать внутри других слов)
    # (...) - группа ключевых слов
    # [ \t]+ - один или более обычных пробелов или табуляций (именно это мы ищем как ошибку)
    # (\d+) - число
    # Примечание: Мы специально ищем обычный пробел. Если там будет \u00A0 (NBSP),
    # регулярное выражение не сработает, что и требуется (так как NBSP - это норма).
    
    pattern_str = r'(?i)\b(рисунок|рис\.?|таблица|табл\.?|от|до|менее|свыше|более)[ \t]+(\d+)'
    return re.compile(pattern_str)

# -------------------------------------------------------------------------
# 2. Реализация функции поиска (Task 2)
# -------------------------------------------------------------------------

def find_missing_nbsp_in_pdf(file_path):
    """
    Ищет в PDF-файле вхождения, где вместо неразрывного пробела стоит обычный.
    
    Аргументы:
        file_path (str): Путь к файлу PDF.
        
    Возвращает:
        tuple: (
            found_status (bool): Истина, если найдены "ошибочные" вхождения, иначе Ложь.
            matches (list): Список кортежей вида ('найденная строка', (номер_страницы, номер_строки)).
        )
    """
    matches_list = []
    regex = get_regex_pattern()
    
    try:
        reader = PdfReader(file_path)
    except Exception as e:
        print(f"Ошибка при открытии файла: {e}")
        return False, []

    # Перебор страниц
    for page_num, page in enumerate(reader.pages):
        # Извлечение текста
        text = page.extract_text()
        
        if not text:
            continue
            
        # Разбиваем текст на строки.
        # В PDF понятие "строка" условно, здесь используем разделитель переноса,
        # который дает PyPDF2.
        lines = text.split('\n')
        
        for line_num, line in enumerate(lines):
            # Поиск всех совпадений в текущей строке
            # finditer позволяет получить объект match с точным текстом
            for match in regex.finditer(line):
                found_str = match.group(0) # Полная найденная строка (например, "рисунок 1")
                
                # Добавляем в результат.
                # page_num + 1 и line_num + 1 для человекочитаемой нумерации (с 1)
                coordinates = (page_num + 1, line_num + 1)
                matches_list.append((found_str, coordinates))

    has_errors = len(matches_list) > 0
    return has_errors, matches_list

# -------------------------------------------------------------------------
# 3. Реализация программного обеспечения (Task 3)
# -------------------------------------------------------------------------

def main():
    print("=== Проверка PDF на отсутствие неразрывных пробелов ===")
    print("Проверяются сочетания: 'рисунок X', 'таблица Y', 'от X', 'менее Y' и т.д.")
    
    # Получение пути к файлу от пользователя

        file_path = sys.argv[1]
    else:
        file_path = input("Введите путь к pdf-файлу: ").strip()
        # Удаляем кавычки, если пользователь скопировал путь как "C:\path..."
        file_path = file_path.replace('"', '').replace("'", "")

    if not os.path.exists(file_path):
        print("Файл не найден.")
        return

    if not file_path.lower().endswith('.pdf'):
        print("Указанный файл не является PDF.")
        return

    print(f"\nАнализ файла: {file_path} ...")
    
    # Вызов функции поиска
    is_found, results = find_missing_nbsp_in_pdf(file_path)

    # Вывод результатов
    if is_found:
        print(f"\n[!] Найдено потенциальных ошибок: {len(results)}")
        print("В следующих местах обнаружен обычный пробел вместо неразрывного:")

if __name__ == "__main__":
```

```python
import re
import sys
import os
from PyPDF2 import PdfReader

# -------------------------------------------------------------------------
# 1. Разработка регулярных выражений (Task 1)
# -------------------------------------------------------------------------

def get_regex_pattern():
    """
    Возвращает скомпилированное регулярное выражение.
    
    Логика:
    1. Ищем ключевое слово (рисунок, таблица, от, до...).
    2. После него ищем ОДИН ИЛИ БОЛЕЕ символов, которые являются "разрываемыми" пробелами.
       Сюда входят: пробел, табуляция (\t), новая строка (\n), возврат каретки (\r),
       вертикальная табуляция (\v), перевод страницы (\f).
    3. Ищем число.
    
    Важно: Мы НЕ используем просто \s, так как в некоторых реализациях \s может 
    включать и неразрывный пробел (\u00A0), а нам нужно найти именно его отсутствие.
    Поэтому мы явно перечисляем "плохие" символы.
    """
    # Список слов
    keywords = r'(рисунок|рис\.?|таблица|табл\.?|от|до|менее|свыше|более)'
    
    # [ \t\n\r\f\v]+ - один или более символов стандартной пустоты (включая перенос строки!)
    bad_spaces = r'[ \t\n\r\f\v]+'
    
    digits = r'(\d+)'
    
    pattern_str = fr'(?i)\b{keywords}{bad_spaces}{digits}'
    return re.compile(pattern_str)

# -------------------------------------------------------------------------
# 2. Реализация функции поиска (Task 2)
# -------------------------------------------------------------------------

def find_missing_nbsp_in_pdf(file_path):
    """
    Ищет в PDF-файле "опасные" разрывы.
    
    Возвращает:
        (status, list_of_matches)
    """
    matches_list = []
    regex = get_regex_pattern()
    
    try:
        reader = PdfReader(file_path)
    except Exception as e:
        print(f"Ошибка при открытии файла: {e}")
        return False, []

    for page_num, page in enumerate(reader.pages):
        text = page.extract_text()
        
        if not text:
            continue

        # Ищем по всему тексту страницы сразу, чтобы поймать разрыв "Рисунок \n 1"
        for match in regex.finditer(text):
            found_str = match.group(0)
            
            # Определяем номер строки.
            # match.start() - индекс начала совпадения в общем тексте страницы.
            # Считаем количество переносов строк (\n) до этого места.
            # +1, так как нумерация строк для человека начинается с 1.
            line_number = text.count('\n', 0, match.start()) + 1
            
            # Для красоты вывода заменяем переносы строк внутри найденного фрагмента на пробел,
            # чтобы в консоли таблица не ломалась.
            display_str = found_str.replace('\n', ' [LF] ').replace('\r', '')
            
            coordinates = (page_num + 1, line_number)
            matches_list.append((display_str, coordinates))

    has_errors = len(matches_list) > 0
    return has_errors, matches_list

# -------------------------------------------------------------------------
# 3. Реализация программного обеспечения (Task 3)
# -------------------------------------------------------------------------

def main():
    print("=== Проверка PDF на отсутствие неразрывных пробелов ===")
    
    if len(sys.argv) > 1:
        file_path = sys.argv[1]
    else:
        file_path = input("Введите путь к pdf-файлу: ").strip()
        file_path = file_path.replace('"', '').replace("'", "")

    if not os.path.exists(file_path):
        print("Файл не найден.")
        return

    print(f"\nАнализ файла: {file_path} ...")
    
    is_found, results = find_missing_nbsp_in_pdf(file_path)

    if is_found:
        print(f"\n[!] Найдено потенциальных ошибок: {len(results)}")
        print("Найдены сочетания с обычными пробелами или уже разорванные переносом строки:")
        print(f"{'Стр.':<5} | {'Строка':<7} | {'Фрагмент (LF = перенос)'}")
        print("-" * 50)
        
        for text_fragment, coords in results:
            page, line = coords
            # Обрезаем слишком длинные фрагменты для красивого вывода
            display_text = (text_fragment[:45] + '..') if len(text_fragment) > 45 else text_fragment
            print(f"{page:<5} | {line:<7} | {display_text}")
    else:
        print("\n[OK] Ошибок не найдено. Везде используются неразрывные пробелы.")

if __name__ == "__main__":
    main()
```

### Почему это решение правильнее:

1.  **Регулярное выражение `[ \t\n\r\f\v]+`**:
    Оно явно нацелено на все символы, которые PyPDF2 (и стандарты ASCII/Unicode) считают разделителями, допускающими перенос строки. Мы намеренно *не* используем `\u00A0` (NBSP), поэтому если регулярное выражение сработало, значит, там стоит что-то, что позволяет разорвать строку.

2.  **Поиск через `finditer(text)` вместо `split`**:
    Если PyPDF2 извлек текст как:
    ```text
    ... см. рисунок
    1. График зависимости...
    ```
    Предыдущий код проверял бы строку "... см. рисунок" отдельно и "1. График..." отдельно. Совпадения бы не было.
    Новый код видит это как единый поток: `...рисунок\n1...`. Регулярное выражение ловит `рисунок` + `\n` + `1`.

3.  **Подсчет строк через `text.count('\n', 0, match.start())`**:
    Это надежный способ узнать, на какой визуальной строке (с точки зрения извлеченного текста) начинается проблемное место, не разбивая сам текст на список.
