3.  **Функция**: Извлекает текст постранично, разбивает его на строки по символу переноса (`\n`) и проверяет каждую строку.

```python
import re
import os
import PyPDF2

def get_validation_regex():
    """
    1. Разработать регулярные выражения для решения задачи по варианту.
    
    Вариант: Проверка отсутствия неразрывного пробела (поиск обычных пробелов)
    в сочетаниях: «рисунок 1», «таблица 2», «от 1», «до 5», «менее 5», «свыше 25».
    
    Логика: Мы ищем "Плохой паттерн" (Слово + Обычный пробел + Число).
    Если паттерн найден, значит, правило нарушено.
    
    (?i) - флаг игнорирования регистра (чтобы находить и "Рис.", и "рис.")
    \b   - граница слова (чтобы не находить внутри других слов)
    (...) - группа ключевых слов
    [ ]+ - один или более ОБЫЧНЫХ пробелов (ASCII 32). Неразрывный пробел сюда не попадет.
    \d+  - одно или более чисел
    """
    pattern = r'(?i)\b(рисунок|рис\.|таблица|табл\.|от|до|менее|свыше|более)[ ]+(\d+)'
    return re.compile(pattern)

def find_formatting_errors(pdf_path):
    """
    2. Функция для поиска подстроки по варианту в pdf-файле.
    
    Аргументы:
        pdf_path (str): Путь к файлу.
        
    Возвращает:
        tuple: (
            bool: True, если найдены ошибки (обычные пробелы вместо неразрывных), иначе False.
            list: Список кортежей [('найденная строка', (стр, строка)), ...]
        )
    """
    regex = get_validation_regex()
    found_errors = []
    
    if not os.path.exists(pdf_path):
        print(f"Ошибка: Файл {pdf_path} не найден.")
        return (False, [])

    try:
        # Открываем PDF файл
        with open(pdf_path, 'rb') as f:
            reader = PyPDF2.PdfReader(f)
            
            # Перебор страниц (нумерация с 0 внутри, для вывода добавим +1)
            for page_idx, page in enumerate(reader.pages):
                text = page.extract_text()
                
                if not text:
                    continue
                
                # Разбиваем текст страницы на строки
                # PyPDF2 обычно сохраняет визуальные строки, разделяя их \n
                lines = text.split('\n')
                
                for line_idx, line in enumerate(lines):
                    # Поиск всех вхождений в строке
                    # finditer возвращает объекты match, из которых можно достать саму строку
                    matches = regex.finditer(line)
                    
                    for match in matches:
                        found_str = match.group(0)
                        # Координаты: (Номер страницы, Номер строки)
                        # Нумерация для пользователя обычно начинается с 1
                        coordinates = (page_idx + 1, line_idx + 1)
                        
                        found_errors.append((found_str, coordinates))
                        
    except Exception as e:
        print(f"Произошла ошибка при чтении PDF: {e}")
        return (False, [])

    # Первый элемент кортежа: True, если список ошибок не пуст
    has_errors = len(found_errors) > 0
    
    return (has_errors, found_errors)

def main():
    """
    3. Реализовать программное обеспечение, принимающее на вход путь к pdf-файлу.
    """
    print("=== Проверка PDF на наличие разрывных пробелов в спец. сочетаниях ===")
    print("Критерии: 'рисунок X', 'таблица X', 'от/до/менее/свыше X' с обычным пробелом.")
    
    file_path = input("Введите путь к PDF-файлу (например, document.pdf): ").strip()
    
    # Убираем кавычки, если пользователь скопировал путь с ними
    file_path = file_path.replace('"', '').replace("'", "")
    
    is_found, results = find_formatting_errors(file_path)
    
    print("-" * 50)
    if is_found:
        print(f"Результат: ИСТИНА (Найдены нарушения форматирования: {len(results)} шт.)")
        print("Список найденных вхождений (текст, (страница, строка)):")
        for item in results:
            text_str = item[0]
            coords = item[1]
            print(f" -> '{text_str}' | Стр. {coords[0]}, Стрк. {coords[1]}")
            
        print("\nРекомендация: Замените обычные пробелы на неразрывные (Ctrl+Shift+Space в Word) в указанных местах.")
    else:
        print("Результат: ЛОЖЬ (Нарушений не найдено или список пуст)")
        print("Форматирование специальных сочетаний корректно (использованы неразрывные пробелы или сочетания отсутствуют).")
    print("-" * 50)

if __name__ == "__main__":
    main()
```
