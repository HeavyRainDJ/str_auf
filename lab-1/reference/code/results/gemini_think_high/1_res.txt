### Описание решения

В коде реализованы следующие этапы:
1.  **Регулярное выражение**: Создан шаблон, который ищет ключевые слова (рисунок, таблица, от, до и т.д.), за которыми следует **обычный** пробел (вместо неразрывного `\u00A0`), а затем цифра.
2.  **Библиотека**: Используется `PyPDF2` (современный класс `PdfReader`).
3.  **Логика поиска**: Текст извлекается постранично, разбивается на строки по символу переноса `\n`. Если в строке находится "запрещенное" сочетание (с обычным пробелом), оно записывается в список результатов.

### Программный код

```python
import re
import os
import PyPDF2

def get_regex_pattern():
    """
    Задача 1: Разработать регулярные выражения для решения задачи по варианту.
    
    Цель: Найти сочетания, где используется обычный пробел вместо неразрывного.
    Ключевые слова: рисунок, рис., таблица, табл., от, до, менее, более, свыше.
    """
    # Список ключевых слов, после которых ожидается число
    keywords = [
        r"рисунок", r"рис\.", 
        r"таблица", r"табл\.", 
        r"от", r"до", 
        r"менее", r"более", r"свыше"
    ]
    
    # Собираем слова в группу (или | или). (?i) - игнорировать регистр.
    keywords_group = "|".join(keywords)
    
    # Регулярное выражение:
    # (?i) - флаг регистронезависимости
    # \b - граница слова (чтобы не искать внутри других слов)
    # (...) - первая группа: одно из ключевых слов
    # [ ]+ - один или несколько ОБЫЧНЫХ пробелов (именно ошибку мы и ищем)
    # (\d+) - вторая группа: одно или несколько чисел
    pattern = re.compile(rf"(?i)\b({keywords_group})[ ]+(\d+)")
    
    return pattern

def analyze_pdf_structure(file_path):
    """
    Задача 2: Реализовать функцию для поиска подстроки по варианту.
    
    Возвращает кортеж:
    - (bool) истина, если найдены ошибки (обычные пробелы там, где нужны неразрывные);
    - (list) список кортежей: [("найденная строка", (номер_страницы, номер_строки)), ...].
    """
    
    regex = get_regex_pattern()
    results = []
    found_any = False
    
    try:
        # Открываем PDF файл
        reader = PyPDF2.PdfReader(file_path)
        
        # Перебор страниц (PyPDF2 нумерует с 0, для пользователя сделаем с 1)
        for page_idx, page in enumerate(reader.pages):
            text = page.extract_text()
            
            if not text:
                continue
            
            # Разбиваем текст страницы на строки
            lines = text.split('\n')
            
            for line_idx, line in enumerate(lines):
                # Поиск всех совпадений в текущей строке
                matches = regex.findall(line)
                
                if matches:
                    found_any = True
                    # matches вернет список кортежей групп, например [('Рисунок', '1'), ('табл.', '5')]
                    # Нам нужно сохранить полную строку и координаты
                    for match in matches:
                        # Восстанавливаем найденную подстроку (слово + пробел + число)
                        found_str = f"{match[0]} {match[1]}"
                        
                        # Координаты: (Номер страницы, Номер строки)
                        # Добавляем +1, так как пользователю привычнее нумерация с 1
                        coordinates = (page_idx + 1, line_idx + 1)
                        
                        results.append((found_str, coordinates))
                        
    except FileNotFoundError:
        print("Ошибка: Файл не найден.")
        return False, []
    except Exception as e:
        print(f"Произошла ошибка при чтении PDF: {e}")
        return False, []

    return found_any, results

def main():
    """
    Задача 3: Реализовать ПО, принимающее на вход путь к pdf-файлу.
    """
    print("=== Анализатор PDF на наличие типографских ошибок (неразрывные пробелы) ===")
    print("Вариант: Поиск отсутствия неразрывного пробела в сочетаниях 'рисунок 1', 'от 5' и т.д.")
    
    file_path = input("Введите путь к PDF-файлу (например, document.pdf): ").strip()
    
    # Убираем кавычки, если пользователь скопировал путь как "C:\Path\..."
    file_path = file_path.replace('"', '')

    if not os.path.exists(file_path):
        print("Файл по указанному пути не существует.")
        return

    # Вызов функции поиска
    is_found, details = analyze_pdf_structure(file_path)

    print("\n--- Результаты анализа ---")
    if is_found:
        print("СТАТУС: Найдены потенциальные ошибки (использован обычный пробел вместо неразрывного).")
        print(f"Всего найдено вхождений: {len(details)}")
        print("\nДетализация:")
        print(f"{'Страница':<10} | {'Строка':<10} | {'Найденный фрагмент'}")
        print("-" * 50)
        
        for item in details:
            text_fragment = item[0]
            coords = item[1]
            page_num = coords[0]
            line_num = coords[1]
            print(f"{page_num:<10} | {line_num:<10} | '{text_fragment}'")
    else:
        print("СТАТУС: Ошибок не обнаружено (либо используется неразрывный пробел, либо сочетания отсутствуют).")

if __name__ == "__main__":
    main()
```

### Пояснения к реализации

*   **Регулярное выражение**: Используется паттерн `[ ]+` (пробел в квадратных скобках), чтобы явно указать поиск стандартного пробела (ASCII 32). Если в документе стоит настоящий неразрывный пробел (NBSP, `\u00A0`), регулярное выражение его пропустит, что и требуется (так как это правильное оформление).
*   **Определение "Строки"**: PDF — это формат фиксированной верстки, в нем нет понятия "строка" как в Word. `PyPDF2` пытается восстановить структуру текста и добавляет символы переноса `\n`. Программа считает строки именно по этим символам переноса в извлеченном тексте.
*   **Координаты**: Возвращаются в формате `(Номер страницы, Порядковый номер строки текста на странице)`.
