```python
import re
import PyPDF2
from typing import Tuple, List, Optional


def create_patterns() -> List[re.Pattern]:
    """
    Создаёт список регулярных выражений для поиска специальных сочетаний слов и чисел,
    где должен использоваться неразрывный пробел.
    """
    patterns = [
        # «рисунок 1», «Рисунок 2» и т.п.
        re.compile(r'[Рр]исунок\s+\d+', re.UNICODE),
        # «таблица 1», «Таблица 2» и т.п.
        re.compile(r'[Тт]аблица\s+\d+', re.UNICODE),
        # «от 1 до 5», «от 10 до 20» и т.п.
        re.compile(r'[Оо]т\s+\d+\s+[Дд]о\s+\d+', re.UNICODE),
        # «менее 5», «Менее 10» и т.п.
        re.compile(r'[Мм]енее\s+\d+', re.UNICODE),
        # «свыше 25», «Свыше 100» и т.п.
        re.compile(r'[Сс]выше\s+\d+', re.UNICODE)
    ]
    return patterns



def find_problematic_spaces(pdf_path: str) -> Tuple[bool, List[Tuple[str, int, int]]:
    """
    Функция для поиска подстрок с обычными пробелами в специальных сочетаниях.
    
    Args:
        pdf_path (str): путь к PDF-файлу
        
    Returns:
        Tuple[bool, List[Tuple[str, int, int]]]: 
            - bool: True, если все найденные сочетания используют неразрывные пробелы (или не найдены),
                   False, если есть сочетания с обычными пробелами
            - List[Tuple[str, int, int]]: список кортежей (найденная строка, номер страницы, номер строки)
              Если bool == True, список пустой
    """
    patterns = create_patterns()
    problematic_occurrences = []
    
    try:
        with open(pdf_path, 'rb') as file:
            pdf_reader = PyPDF2.PdfReader(file)
            
            for page_num in range(len(pdf_reader.pages)):
                page = pdf_reader.pages[page_num]
                text = page.extract_text()
                
                if not text:
                    continue
                
                # Разбиваем текст на строки
                lines = text.splitlines()
                
                for line_num, line in enumerate(lines, start=1):
                    # Проверяем каждую строку на соответствие шаблонам
                    for pattern in patterns:
                        matches = pattern.finditer(line)
                        for match in matches:
                            found_text = match.group()
                            # Проверяем, есть ли в найденном тексте обычный пробел (не неразрывный)
                            if ' ' in found_text:  # обычный пробел
                                problematic_occurrences.append((found_text, page_num + 1, line_num))
    
    except Exception as e:
        print(f"Ошибка при обработке PDF-файла: {e}")
        return False, []
    
    # Если нашли проблемные вхождения — возвращаем False и список
    if problematic_occurrences:
        return False, problematic_occurrences
    
    # Если всё хорошо (проблем нет) — возвращаем True и пустой список
    return True, []



def main(pdf_path: str):
    """
    Основная функция для запуска проверки PDF-файла.
    
    Args:
        pdf_path (str): путь к PDF-файлу для проверки
    """
    print(f"Начинаем проверку файла: {pdf_path}")
    
    has_no_problems, occurrences = find_problematic_spaces(pdf_path)
    
    if has_no_problems:
        print("✅ Все специальные сочетания содержат неразрывные пробелы (или не найдены).")
    else:
        print("❌ Обнаружены сочетания с обычными пробелами (должны быть неразрывными):")
        for text, page, line in occurrences:
            print(f"  - '{text}' (страница {page}, строка {line})")



# Пример использования
if __name__ == "__main__":
    # Укажите путь к вашему PDF-файлу
    pdf_file_path = "example.pdf"
    main(pdf_file_path)
```

### Описание кода

1. **Функция `create_patterns()`**  
   Создаёт список регулярных выражений для поиска целевых сочетаний:
   - «рисунок N», «таблица N»;
   - «от N до M»;
   - «менее N»;
   - «свыше N».  
   Учитываются варианты с заглавной и строчной буквы.

2. **Функция `find_problematic_spaces()`**  
   - Принимает путь к PDF‑файлу.
   - Извлекает текст с каждой страницы через `PyPDF2`.
   - Разбивает текст на строки.
   - Для каждой строки проверяет соответствие шаблонам.
   - Если найдено сочетание с обычным пробелом (а не неразрывным) — добавляет в список проблем.
   - Возвращает:
     - `True`, если проблем нет (или сочетания не найдены);
     - `False` и список проблемных вхождений, если найдены обычные пробелы.

3. **Функция `main()`**  
   - Запускает проверку и выводит результат в понятном виде:
     - ✅ если всё хорошо;
     - ❌ и список проблемных мест, если найдены ошибки.

### Примечания

- Код ищет **обычные пробелы** в целевых сочетаниях. Неразрывные пробелы (например, `\xa0` в Unicode) не будут считаться проблемой.
- Если PDF содержит только изображения или плохо распознанный текст, результат может быть неточным.
- Для более точного распознавания можно доработать код с использованием `pdfplumber` или `pytesseract` (для OCR).
