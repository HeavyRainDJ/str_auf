```python
import re
import PyPDF2
from typing import Tuple, List, Optional


def create_patterns() -> List[str]:
    """
    Создаёт список регулярных выражений для поиска специальных сочетаний слов и чисел,
    где требуется неразрывный пробел.
    """
    patterns = [
        r'рисунок\s+\d+',      # «рисунок 1», «рисунок 25» и т.п.
        r'таблица\s+\d+',      # «таблица 2», «таблица 10» и т.п.
        r'от\s+\d+\s+до\s+\d+', # «от 1 до 5», «от 10 до 20» и т.п.
        r'менее\s+\d+',         # «менее 5», «менее 100» и т.п.
        r'свыше\s+\d+'          # «свыше 25», «свыше 1000» и т.п.
    ]
    return patterns



def find_issues_in_pdf(pdf_path: str) -> Tuple[bool, List[Tuple[str, int, int]]:
    """
    Функция для поиска проблемных мест (отсутствие неразрывного пробела) в PDF-файле.

    Args:
        pdf_path (str): Путь к PDF-файлу.

    Returns:
        Tuple[bool, List[Tuple[str, int, int]]]:
            - bool: True, если все найденные сочетания корректны (есть неразрывный пробел),
                     False, если найдены проблемные места (нет неразрывного пробела).
            - List[Tuple[str, int, int]]: Список кортежей (найденная строка, номер страницы, номер строки),
                                            если проблем нет — пустой список.
    """
    patterns = create_patterns()
    issues = []  # Список проблемных мест

    try:
        with open(pdf_path, 'rb') as file:
            pdf_reader = PyPDF2.PdfReader(file)
            total_pages = len(pdf_reader.pages)

            for page_num in range(total_pages):
                page = pdf_reader.pages[page_num]
                text = page.extract_text()

                if not text:
                    continue

                # Разбиваем текст на строки
                lines = text.splitlines()

                for line_num, line in enumerate(lines, start=1):
                    # Проверяем каждую строку на соответствие шаблонам
                    for pattern in patterns:
                        matches = re.finditer(pattern, line, re.IGNORECASE)
                        for match in matches:
                            found_text = match.group()
                            # Проверяем, есть ли неразрывный пробел (символ \xa0) в найденном фрагменте
                            if '\xa0' not in found_text:
                                issues.append((found_text, page_num + 1, line_num))

    except Exception as e:
        print(f"Ошибка при обработке PDF-файла: {e}")
        return False, []

    # Если issues пуст — все хорошо, неразрывные пробелы на месте
    is_ok = len(issues) == 0
    return is_ok, issues



def main(pdf_path: str):
    """
    Основная функция — принимает путь к PDF-файлу и выводит результат проверки.

    Args:
        pdf_path (str): Путь к PDF-файлу для проверки.
    """
    print(f"Начинаем проверку PDF-файла: {pdf_path}")
    is_ok, issues = find_issues_in_pdf(pdf_path)

    if is_ok:
        print("✅ Все специальные сочетания содержат неразрывный пробел. Проблем не обнаружено.")
    else:
        print("❌ Найденные проблемы (отсутствие неразрывного пробела):")
        for text, page, line in issues:
            print(f"  - '{text}' (страница {page}, строка {line})")



# Пример использования
if __name__ == "__main__":
    # Укажите путь к вашему PDF-файлу
    pdf_file_path = "example.pdf"  # замените на реальный путь
    main(pdf_file_path)
```

### Описание кода

1. **Функция `create_patterns()`**  
   - Формирует список регулярных выражений для поиска целевых сочетаний:
     - `рисунок\s+\d+` — ищет «рисунок» + пробел + число;
     - `таблица\s+\d+` — аналогично для «таблица»;
     - `от\s+\d+\s+до\s+\d+` — шаблон «от X до Y»;
     - `менее\s+\d+` и `свыше\s+\d+` — для соответствующих конструкций.

2. **Функция `find_issues_in_pdf()`**  
   - Принимает путь к PDF-файлу.
   - Открывает файл через `PyPDF2`.
   - Для каждой страницы извлекает текст и разбивает его на строки.
   - В каждой строке ищет совпадения по шаблонам.
   - Если найдено сочетание **без неразрывного пробела** (`\xa0`), добавляет его в список проблем.
   - Возвращает:
     - `True`, если проблем нет;
     - список кортежей `(найденный текст, номер страницы, номер строки)` с проблемами.

3. **Функция `main()`**  
   - Вызывает проверку и выводит результат в удобном виде:
     - Если всё хорошо — сообщение с галочкой.
     - Если найдены проблемы — список проблемных мест с указанием страницы и строки.

### Примечания

- Неразрывный пробел в Unicode — символ `\xa0`.
- Поиск регистронезависимый (`re.IGNORECASE`).
- Если PDF содержит только изображения или плохо распознанный текст, результат может быть неточным.
