import re
import PyPDF2

def find_non_breaking_spaces(pdf_path):
    # Открываем PDF-файл
    with open(pdf_path, 'rb') as file:
        reader = PyPDF2.PdfReader(file)
        num_pages = len(reader.pages)

    # Регулярное выражение для поиска сочетаний «рисунок 1», «таблица 2» и т. д.
    pattern = re.compile(r'\b(рисунок|таблица|от\s+\d+\s+до\s+\d+|менее\s+\d+|свыше\s+\d+)\s+(\d+)\b')

    result = {
        'наличие_неразрывного_пробела': False,
        'найденные_строки': []
    }

    for page_num in range(num_pages):
        page = reader.pages[page_num]
        text = page.extract_text()

        if text:
            for match in pattern.finditer(text):
                found_string = match.group(0)
                # Проверяем наличие неразрывного пробела (Unicode: \u00A0)
                if '\u00A0' in found_string:
                    result['наличие_неразрывного_пробела'] = True
                # Определяем номер строки, в которой найдено вхождение
                line_num = text[:match.start()].count('\n') + 1
                result['найденные_строки'].append((found_string, (page_num + 1, line_num)))

    return (result['наличие_неразрывного_пробела'], result['найденные_строки'])

def main():
    pdf_path = input("Введите путь к PDF-файлу: ")
    result = find_non_breaking_spaces(pdf_path)
    print("Наличие неразрывного пробела:", result[0])
    if result[1]:
        print("Найденные строки:")
        for item in result[1]:
            print(f"Строка: {item[0]}, координаты: страница {item[1][0]}, строка {item[1][1]}")
    else:
        print("Строки не найдены.")

if __name__ == "__main__":
    main()

import re
import PyPDF2

def find_missing_non_breaking_spaces(pdf_path):
    """
    Проверяет отсутствие неразрывных пробелов (\u00A0) в ключевых сочетаниях:
    - «рисунок 1»
    - «таблица 2»
    - «от 1 до 5»
    - «менее 5»
    - «свыше 25»

    Возвращает:
    (есть_ошибки: bool, список_находок: list)
    где каждая находка — (строка, (страница, строка_в_тексте))
    """
    # Шаблоны для поиска ключевых сочетаний
    patterns = [
        r'рисунок\s+\d+',          # «рисунок 1», «рисунок 10»
        r'таблица\s+\d+',          # «таблица 2», «таблица 5»
        r'от\s+\d+\s+до\s+\d+',  # «от 1 до 5», «от 10 до 20»
        r'менее\s+\d+',            # «менее 5», «менее 100»
        r'свыше\s+\d+'             # «свыше 25», «свыше 1000»
    ]

    compiled_patterns = [re.compile(p, re.IGNORECASE) for p in patterns]

    errors = []  # Список находок, где НЕТ неразрывного пробела

    try:
        with open(pdf_path, 'rb') as file:
            reader = PyPDF2.PdfReader(file)
            num_pages = len(reader.pages)

            for page_idx in range(num_pages):
                page = reader.pages[page_idx]
                text = page.extract_text()

                if not text:
                    continue

                # Разбиваем текст на строки для подсчёта номера строки
                lines = text.splitlines()

                for line_idx, line in enumerate(lines, start=1):
                    for pattern in compiled_patterns:
                        for match in pattern.finditer(line):
                            matched_text = match.group(0)

                            # Проверяем: есть ли в найденном фрагменте неразрывный пробел (\u00A0)?
                            if '\u00A0' not in matched_text:
                                # ОШИБКА: неразрывного пробела нет!
                                errors.append((
                                    matched_text,
                                    (page_idx + 1, line_idx)  # (страница, номер строки)
                                ))
    except Exception as e:
        raise RuntimeError(f"Ошибка при обработке PDF: {e}")

    # Если errors не пуст — значит, есть нарушения (не хватает неразрывных пробелов)
    has_errors = len(errors) > 0

    return (has_errors, errors)



def main():
    pdf_path = input("Введите путь к PDF-файлу: ").strip()
    
    try:
        has_errors, findings = find_missing_non_breaking_spaces(pdf_path)

        print("\n! + "="*60)
        if has_errors:
            print("ОБНАРУЖЕНЫ ОШИБКИ: отсутствуют неразрывные пробелы")
            print(f"Всего нарушений: {len(findings)}")
            print("-! * 60)
            for text, (page, line) in findings:
                print(f"→ '{text}' | страница {page}, строка {line}")
        else:
            print("Все ключевые сочетания содержат неразрывные пробелы. Ошибок нет.")

    except FileNotFoundError:
        print("Файл не найден. Проверьте путь.")
    except Exception as e:
        print(f"Ошибка: {e}")



if __name__ == "__main__":
    main()

import re
import PyPDF2

def find_non_breaking_spaces(pdf_path):
    with open(pdf_path, 'rb') as file:
        reader = PyPDF2.PdfReader(file)
        num_pages = len(reader.pages)
        
        # Регулярное выражение для поиска сочетаний «рисунок 1», «таблица 2», «рис. 1», «таб. 2», «от 1 до 5», «менее 5», «свыше 25»
        pattern = re.compile(r'\b(рисунок\s+\d+|таблица\s+\d+|рис\.\s+\d+|таб\.\s+\d+|от\s+\d+|до\s+\d+|менее\s+\d+|свыше\s+\d+)\b', re.IGNORECASE)
        result = {
            'наличие_ошибки_с_пробелом': False,
            'найденные_строки': []
        }
        
        for page_num in range(num_pages):
            page = reader.pages[page_num]
            text = page.extract_text()
            if text:
                for match in pattern.finditer(text):
                    found_string = match.group(0)
                    # Проверяем отсутствие неразрывного пробела (Unicode: \u00A0)
                    if '\u00A0' not in found_string:
                        result['наличие_ошибки_с_пробелом'] = True
                    
                    # Определяем номер строки, в которой найдено вхождение
                    line_num = text[:match.start()].count('\n') + 1
                    result['найденные_строки'].append((found_string, (page_num + 1, line_num)))
    
    return (result['наличие_ошибки_с_пробелом'], result['найденные_строки'])

def main():
