```python
import re
import os
try:
    from pypdf import PdfReader
except ImportError:
    try:
        from PyPDF2 import PdfReader
    except ImportError:
        print("Ошибка: Библиотека pypdf или PyPDF2 не найдена. Установите её: pip install pypdf")
        exit()

def get_regex_pattern():
    """
    Задача 1: Разработать регулярные выражения для решения задачи по варианту.
    
    Вариант: Проверка отсутствия неразрывного пробела.
    Мы ищем обычный пробел (\x20 или просто " ") там, где должен быть неразрывный (\xA0).
    
    Ключевые слова: рисунок, рис., таблица, табл., от, до, менее, свыше, более.
    После них следует число.
    """
    # Список слов-триггеров (можно расширять). (?i) - флаг игнорирования регистра
    # \b - граница слова, чтобы не ловить "подо" как "до"
    keywords = [
        r"рисунок", r"рис\.", 
        r"таблица", r"табл\.", 
        r"от", r"до", 
        r"менее", r"свыше", r"более"
    ]
    
    keywords_pattern = "|".join(keywords)
    
    # Регулярное выражение:
    # (?i) - игнорировать регистр
    # \b(...) - начало слова и группа вариантов
    # [ ] - ищем именно ОБЫЧНЫЙ пробел (ошибка верстки)
    # \d+ - одно или более чисел
    pattern = rf"(?i)\b({keywords_pattern})[ ]\d+"
    
    return pattern

def search_in_pdf(pdf_path):
    """
    Задача 2: Реализовать функцию для поиска подстроки по варианту.
    
    Возвращает кортеж:
    - (bool): Истина, если найдена хотя бы одна ошибка (обычный пробел вместо неразрывного).
    - (list): Список кортежей (найденная_строка, (номер_страницы, приблизительный_номер_строки)).
    """
    regex = get_regex_pattern()
    compiled_regex = re.compile(regex)
    
    results = []
    found_any = False
    
    try:
        reader = PdfReader(pdf_path)
        
        # Перебираем страницы
        for page_num, page in enumerate(reader.pages):
            # Извлекаем текст. extract_text() возвращает текст с переносами строк \n
            text = page.extract_text()
            
            if not text:
                continue
            
            # Разбиваем текст на строки для определения "номера строки документа"
            # Примечание: В PDF понятие "строка" условно, extract_text делает приближение.
            lines = text.split('\n')
            
            for line_num, line in enumerate(lines):
                # Ищем совпадения в конкретной строке
                matches = compiled_regex.finditer(line)
                
                for match in matches:
                    found_any = True
                    match_str = match.group()
                    # Формируем координаты: (Номер страницы + 1, Номер строки + 1) для удобства чтения человеком
                    coords = (page_num + 1, line_num + 1)
                    results.append((match_str, coords))
                    
    except Exception as e:
        print(f"Произошла ошибка при чтении файла: {e}")
        return False, []

    return found_any, results

def main():
    """
    Задача 3: Реализовать программное обеспечение, принимающее путь к PDF.
    """
    print("=== Анализатор PDF на наличие ошибочных обычных пробелов ===")
    print("Ищет сочетания типа 'рис. 1', 'от 5' с обычным пробелом вместо неразрывного.")
    
    # Запрос пути к файлу
    file_path = input("Введите путь к файлу (например, document.pdf): ").strip()
    
    # Удаляем кавычки, если пользователь скопировал путь как "C:\Path\..."
    if file_path.startswith('"') and file_path.endswith('"'):
        file_path = file_path[1:-1]
        
    if not os.path.exists(file_path):
        print("Ошибка: Файл не найден.")
        return

    if not file_path.lower().endswith('.pdf'):
        print("Ошибка: Файл должен иметь расширение .pdf")
        return

    print(f"\nАнализ файла: {file_path} ...")
    
    # Вызов функции поиска
    has_errors, error_list = search_in_pdf(file_path)
    
    print("\n=== Результаты анализа ===")
    if has_errors:
        print(f"Найдены потенциальные ошибки (обычный пробел вместо неразрывного): {len(error_list)} шт.")
        print("-" * 60)
        print(f"{'Найдено':<30} | {'Стр.':<5} | {'Строка':<5}")
        print("-" * 60)
        for text, coords in error_list:
            page, line = coords
            print(f"'{text}':<30 | {page:<5} | {line:<5}")
        print("-" * 60)
        print("РЕЗЮМЕ: Истина (найдены нарушения правила неразрывного пробела)")
    else:
        print("Ошибок не найдено. Во всех ключевых сочетаниях используются либо неразрывные пробелы, либо сочетания отсутствуют.")
        print("РЕЗЮМЕ: Ложь")

if __name__ == "__main__":
    main()
```

### Пояснения к коду

1.  **Регулярное выражение (`get_regex_pattern`):**
    *   Мы формируем список ключевых слов: `рис.`, `таблица`, `от`, `до` и т.д.
    *   `rf"(?i)\b({keywords_pattern})[ ]\d+"`
        *   `(?i)` — делает поиск нечувствительным к регистру (найдет и "Рисунок", и "рисунок").
        *   `\b` — граница слова. Это важно, чтобы слово "под**от**чет" не сработало как предлог "от".
        *   `[ ]` — явно ищет обычный пробел. Если в PDF стоит `\xa0` (non-breaking space), регулярка его **не** найдет (что нам и нужно, так как мы ищем именно ошибки).
        *   `\d+` — цифры, идущие после пробела.

2.  **Поиск в PDF (`search_in_pdf`):**
    *   Используется `PdfReader` для открытия файла.
    *   `page.extract_text()` извлекает текстовый слой.
    *   Текст разбивается по символу переноса строки `\n`, чтобы мы могли посчитать номер строки, как того требует задание.
    *   Мы ищем вхождения регулярного выражения в каждой строке.
    *   Результат накапливается в список `results`.

3.  **Интерфейс (`main`):**
    *   Просит пользователя ввести путь.
    *   Проверяет существование файла.
    *   Выводит таблицу с результатами: текст ошибки и координаты (страница, строка).
