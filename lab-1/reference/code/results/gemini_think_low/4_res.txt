```python
import re
import os
from PyPDF2 import PdfReader

def get_regex_patterns():
    """
    Задача 1: Разработать регулярные выражения для решения задачи по варианту.
    Вариант: Поиск отсутствия неразрывного пробела в сочетаниях слов и чисел.
    Мы ищем "ошибочные" паттерны, где стоит обычный пробел вместо неразрывного.
    """
    
    # \s - любой пробельный символ (включая обычный пробел, таб, перенос строки), 
    # но нам нужно исключить случай, если там уже стоит неразрывный пробел (\u00A0).
    # Однако PyPDF2 при извлечении текста часто заменяет все пробелы на обычные,
    # либо оставляет их как есть.
    # Для целей проверки мы будем искать именно ОБЫЧНЫЙ пробел (код 32) или перенос строки
    # в критических местах.
    
    # Паттерн 1: "рисунок X", "таблица Y" (регистронезависимо)
    # Пример: "Рисунок 1", "табл. 5"
    p1 = r'(?:рисунок|рис\.|таблица|табл\.)\s+\d+'
    
    # Паттерн 2: Диапазоны и сравнения "от X", "до Y", "менее Z", "свыше N", "более M"
    # Пример: "от 1", "до 50", "менее 5"
    p2 = r'(?:от|до|менее|свыше|более)\s+\d+'
    
    # Объединяем в один общий паттерн с флагом IGNORECASE
    # Мы ищем именно обычный пробел (\x20) или перенос строки (\n, \r), 
    # так как PyPDF2 обычно декодирует текст в UTF-8.
    # Важно: Неразрывный пробел в Python это \u00A0. 
    # Мы ищем: (Слово)[Обычный Пробел](Цифра)
    
    keywords = r'(?:рисунок|рис\.|таблица|табл\.|от|до|менее|свыше|более)'
    
    # Регулярное выражение ищет: Ключевое слово + Обычный пробел ( ) + Цифра
    # Если мы находим это, значит неразрывного пробела там нет (ошибка).
    regex_pattern = re.compile(rf'({keywords} \d+)', re.IGNORECASE)
    
    return regex_pattern

def check_pdf_for_missing_nbsp(file_path):
    """
    Задача 2: Реализовать функцию для поиска подстроки по варианту в pdf-файле.
    
    Аргументы:
        file_path (str): Путь к файлу.
        
    Возвращает:
        tuple: (bool, list of tuples)
        - bool: True, если найдены ошибки (отсутствие неразрывного пробела), иначе False.
        - list: Список кортежей [('найденная строка', 'страница: X, строка: Y')].
    """
    
    if not os.path.exists(file_path):
        print(f"Ошибка: Файл {file_path} не найден.")
        return False, []

    pattern = get_regex_patterns()
    results = []
    
    try:
        reader = PdfReader(file_path)
        
        # Проходим по всем страницам
        for page_num, page in enumerate(reader.pages):
            # Извлекаем текст. PyPDF2 не всегда идеально сохраняет структуру строк,
            # но extract_text() пытается вернуть текст с переносами \n
            text = page.extract_text()
            
            if not text:
                continue

            # Разбиваем текст страницы на строки, чтобы определить номер строки
            lines = text.split('\n')
            
            for line_num, line in enumerate(lines):
                # Ищем совпадения в конкретной строке
                matches = pattern.findall(line)
                
                for match in matches:
                    # Формируем кортеж согласно заданию:
                    # (найденная строка, координаты)
                    # Координаты: Стр (начиная с 1), Строка (начиная с 1)
                    coord_info = f"Страница: {page_num + 1}, Строка: {line_num + 1}"
                    results.append((match, coord_info))
                    
    except Exception as e:
        print(f"Произошла ошибка при чтении PDF: {e}")
        return False, []

    # Возвращаем результат
    has_errors = len(results) > 0
    return has_errors, results

def main():
    """
    Задача 3: Реализовать ПО, принимающее на вход путь к pdf-файлу.
    """
    print("=== Проверка PDF на отсутствие неразрывных пробелов ===")
    print("Проверяются сочетания: 'рисунок X', 'таблица Y', 'от/до/менее/свыше Z'")
    
    file_path = input("Введите путь к файлу (например, document.pdf): ").strip()
    
    # Удаляем кавычки, если пользователь скопировал путь с ними
    file_path = file_path.replace('"', '').replace("'", "")
    
    is_found, findings = check_pdf_for_missing_nbsp(file_path)
    
    print("\n" + "="*50)
    print(f"Результат анализа файла: {file_path}")
    print("="*50)
    
    # Согласно заданию: значение, определяющее наличие строки ("истина/ложь")
    print(f"Наличие ошибок (обычный пробел вместо неразрывного): {'ИСТИНА' if is_found else 'ЛОЖЬ'}")
    
    if is_found:
        print(f"\nНайденные вхождения ({len(findings)} шт.):")
        print(f"{'ФРАГМЕНТ':<30} | {'КООРДИНАТЫ'}")
        print("-" * 60)
        for fragment, coords in findings:
            print(f"{fragment:<30} | {coords}")
    else:
        print("\nОшибок не обнаружено. Возможно, везде стоят неразрывные пробелы или таких сочетаний нет.")

if __name__ == "__main__":
    main()
```

### Пояснения к коду:

1.  **Задача 1 (Регулярные выражения):**
    *   В функции `get_regex_patterns` создается регулярное выражение.
    *   Я использовал конструкцию `rf'({keywords} \d+)'`. Обратите внимание на пробел внутри строки после `{keywords}`. Это обычный пробел (ASCII 32).
    *   Если в PDF файле используется настоящий неразрывный пробел (`\u00A0`), регулярное выражение не сработает (что правильно, так как это не ошибка).
    *   Если в PDF файле стоит обычный пробел (ошибка верстки), регулярное выражение найдет это совпадение.
    *   Ключевые слова: *рисунок, рис., таблица, табл., от, до, менее, свыше, более*.

2.  **Задача 2 (Функция поиска):**
    *   Используется `PdfReader` из `PyPDF2`.
    *   Текст извлекается постранично (`extract_text()`).
    *   Текст страницы разбивается по символу переноса строки `\n` на список строк (`lines`), чтобы мы могли выполнить требование задания: указать номер строки.
    *   Результат накапливается в список `results`.
    *   Возвращается кортеж: `(True/False, список_кортежей)`.

3.  **Задача 3 (Интерфейс):**
    *   Функция `main` запрашивает путь к файлу у пользователя.
    *   Выводит результат в читаемом виде.
