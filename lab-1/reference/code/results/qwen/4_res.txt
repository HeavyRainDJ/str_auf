```python
import PyPDF2
import re
from typing import Tuple, List, Optional

def find_special_combinations(pdf_path: str) -> Tuple[bool, List[Tuple[str, int, int]]]:
    """
    Ищет специальные сочетания слов и чисел в PDF-файле,
    проверяя отсутствие неразрывного пробела.
    
    Args:
        pdf_path (str): Путь к PDF-файлу
        
    Returns:
        Tuple[bool, List[Tuple[str, int, int]]]: 
        - bool: True если найдены нарушения, False если все корректно
        - List[Tuple[str, int, int]]: Список кортежей (найденная строка, номер страницы, номер строки)
    """
    
    # Регулярное выражение для поиска специальных сочетаний без неразрывного пробела
    # Совпадает с шаблонами: "рисунок X", "таблица X", "от X до Y", "менее X", "свыше X"
    pattern = r'(?:рисунок|таблица|от\s+\d+\s+до\s+\d+|менее\s+\d+|свыше\s+\d+)(?:\s*\S*)?'
    
    # Регулярное выражение для поиска сочетаний с неразрывным пробелом
    # Ищем последовательности символов, где между словами есть неразрывный пробел
    non_breaking_space_pattern = r'(?:рисунок|таблица|от\s+\d+\s+до\s+\d+|менее\s+\d+|свыше\s+\d+)\s*[\u00A0\u2000-\u200B]\s*\d+'
    
    # Регулярное выражение для поиска корректных сочетаний (без неразрывного пробела)
    correct_pattern = r'(?:рисунок|таблица|от\s+\d+\s+до\s+\d+|менее\s+\d+|свыше\s+\d+)\s+\d+(?:\s*\S*)?'
    
    # Найденные нарушения
    violations = []
    
    try:
        with open(pdf_path, 'rb') as file:
            pdf_reader = PyPDF2.PdfReader(file)
            
            # Проходим по каждой странице
            for page_num, page in enumerate(pdf_reader.pages, 1):
                text = page.extract_text()
                
                # Разбиваем текст на строки
                lines = text.split('\n')
                
                # Ищем все вхождения специальных сочетаний
                for line_num, line in enumerate(lines, 1):
                    # Ищем сочетания с неразрывным пробелом
                    matches = re.finditer(non_breaking_space_pattern, line, re.IGNORECASE)
                    
                    for match in matches:
                        # Проверяем, что это действительно нарушение (найден неразрывный пробел)
                        found_text = match.group()
                        violations.append((found_text, page_num, line_num))
                        
        # Возвращаем результат: True если есть нарушения, False если нет
        return (len(violations) > 0, violations)
        
    except Exception as e:
        print(f"Ошибка при обработке PDF-файла: {e}")
        return (False, [])
```

```python
import PyPDF2
import re
from typing import Tuple, List

def find_special_combinations(pdf_path: str) -> Tuple[bool, List[Tuple[str, int, int]]]:
    """
    Ищет специальные сочетания слов и чисел в PDF-файле,
    проверяя отсутствие неразрывного пробела.
    
    Args:
        pdf_path (str): Путь к PDF-файлу
        
    Returns:
        Tuple[bool, List[Tuple[str, int, int]]]: 
        - bool: True если найдены нарушения, False если все корректно
        - List[Tuple[str, int, int]]: Список кортежей (найденная строка, номер страницы, номер строки)
    """
    
    # Регулярное выражение для поиска специальных сочетаний
    # Ищем: рисунок 1, таблица 2, от 1 до 5, менее 5, свыше 25
    pattern = r'(?:рисунок|таблица|от\s+\d+\s+до\s+\d+|менее\s+\d+|свыше\s+\d+)'
    
    # Регулярное выражение для поиска нарушений (неразрывный пробел)
    # Неразрывные пробелы в Unicode: \u00A0 (NBSP), \u2000-\u200B (различные пробельные символы)
    violation_pattern = r'(?:рисунок|таблица|от\s+\d+\s+до\s+\d+|менее\s+\d+|свыше\s+\d+)\s*[\u00A0\u2000-\u200B]\s*\d+'
    
    # Найденные нарушения
    violations = []
    
    try:
        with open(pdf_path, 'rb') as file:
            pdf_reader = PyPDF2.PdfReader(file)
            
            # Проходим по каждой странице
            for page_num, page in enumerate(pdf_reader.pages, 1):
                text = page.extract_text()
                
                # Разбиваем текст на строки
                lines = text.split('\n')
                
                # Ищем все вхождения с неразрывным пробелом
                for line_num, line in enumerate(lines, 1):
                    # Ищем нарушения (с неразрывным пробелом)
                    matches = re.finditer(violation_pattern, line, re.IGNORECASE)
                    
                    for match in matches:
                        found_text = match.group()
                        violations.append((found_text, page_num, line_num))
                        
        # Возвращаем результат: True если есть нарушения, False если нет
        return (len(violations) > 0, violations)
        
    except Exception as e:
        print(f"Ошибка при обработке PDF-файла: {e}")
        return (False, [])

def analyze_pdf_for_non_breaking_spaces(pdf_path: str) -> None:
    """
    Основная функция для анализа PDF-файла

# Пример использования
if __name__ == "__main__":
```

**Описание работы программы:**

1. **Регулярные выражения:**
   - `pattern`: Ищет специальные сочетания слов и чисел
   - `violation_pattern`: Ищет нарушения с неразрывным пробелом

2. **Функция `find_special_combinations`:**
   - Принимает путь к PDF-файлу
   - Извлекает текст со всех страниц
   - Разбивает текст на строки
   - Ищет сочетания с неразрывным пробелом
   - Возвращает кортеж: (наличие нарушений, список найденных нарушений)

3. **Функция `analyze_pdf_for_non_breaking_spaces`:**
   - Основная функция для анализа
   - Выводит результаты в читаемом формате
