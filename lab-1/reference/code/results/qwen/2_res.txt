```python
import PyPDF2
import re

def check_non_breaking_space_in_combinations(pdf_path):
    """
    Проверяет, отсутствует ли неразрывный пробел в специальных сочетаниях слов и чисел.
    
    :param pdf_path: путь к PDF-файлу
    :return: кортеж (bool, list of tuples)
             - bool: True, если найдены нарушения (неразрывный пробел), иначе False
             - list of tuples: [(найденная строка, номер страницы, номер строки)]
    """
    # Регулярное выражение для поиска специальных сочетаний с возможным неразрывным пробелом
    pattern = re.compile(
        r'(?:рисунок|таблица|от|менее|свыше)\s*[\xa0\s]*\d+(?:\s*до\s*\d+)?',
        re.IGNORECASE
    )

    found_issues = []
    page_num = 0

    try:
        with open(pdf_path, 'rb') as file:
            reader = PyPDF2.PdfReader(file)
            for page in reader.pages:
                page_num += 1
                text = page.extract_text()
                lines = text.splitlines()

                for line_num, line in enumerate(lines, start=1):
                    matches = pattern.findall(line)
                    for match in matches:
                        # Проверяем, есть ли в найденной строке неразрывный пробел
                        if '\xa0' in match:
                            found_issues.append((match, page_num, line_num))

    except Exception as e:
        print(f"Ошибка при чтении PDF: {e}")
        return False, []

    # Возвращаем результат
    return len(found_issues) > 0, found_issues


def main():
    pdf_path = input("Введите путь к PDF-файлу: ").strip()
    has_issues, issues = check_non_breaking_space_in_combinations(pdf_path)

    if has_issues:
        print("❌ Найдены нарушения: неразрывный пробел в сочетаниях слов и чисел.")
        for issue in issues:
            print(f"  - '{issue[0]}' на странице {issue[1]}, строка {issue[2]}")
    else:
        print("✅ Неразрывные пробелы отсутствуют в указанных сочетаниях.")


if __name__ == "__main__":
    main()
```

### 🔍 Объяснение:
1. **Регулярное выражение**:
   ```python
   r'(?:рисунок|таблица|от|менее|свыше)\s*[\xa0\s]*\d+(?:\s*до\s*\d+)?'
   ```
   - Находит слова: `рисунок`, `таблица`, `от`, `менее`, `свыше`
   - Позволяет пробел или неразрывный пробел (`\xa0`) между словом и числом
   - Также учитывает "от X до Y"

2. **Функция `check_non_breaking_space_in_combinations`**:
   - Читает PDF постранично
   - Извлекает текст и разбивает его на строки
   - Проверяет каждую строку с помощью регулярного выражения
   - Если в найденной строке есть `\xa0`, добавляет её в список `found_issues`

3. **Вывод**:
   - Программа выводит результат: есть ли нарушения и где они находятся
