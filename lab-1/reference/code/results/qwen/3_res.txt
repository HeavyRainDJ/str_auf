```python
import PyPDF2
import re
from typing import Tuple, List, Optional

def find_special_combinations(pdf_path: str) -> Tuple[bool, List[Tuple[str, int, int]]]:
    """
    Ищет специальные сочетания слов и чисел в PDF-файле, проверяя отсутствие неразрывного пробела.
    
    Args:
        pdf_path (str): Путь к PDF-файлу
        
    Returns:
        Tuple[bool, List[Tuple[str, int, int]]]: 
        - bool: True если найдены нарушения, False если нет
        - List[Tuple[str, int, int]]: Список кортежей (найденная строка, номер страницы, номер строки)
    """
    
    # Регулярное выражение для поиска специальных сочетаний без неразрывного пробела
    # Сочетания: "рисунок 1", "таблица 2", "от 1 до 5", "менее 5", "свыше 25"
    pattern = r'(?:рисунок|таблица|от\s+\d+\s+до\s+\d+|менее\s+\d+|свыше\s+\d+)(?:\s*\S*)?'
    
    # Регулярное выражение для поиска сочетаний с неразрывным пробелом (для проверки)
    pattern_with_nbsp = r'(?:рисунок|таблица|от\s+\d+\s+до\s+\d+|менее\s+\d+|свыше\s+\d+)(?:\s*[\xa0]\S*)?'
    
    # Проверяем на наличие неразрывных пробелов в специальных сочетаниях
    # Это регулярное выражение ищет сочетания, где между словом и числом есть неразрывный пробел
    nbsp_pattern = r'(?:рисунок|таблица|от\s+\d+\s+до\s+\d+|менее\s+\d+|свыше\s+\d+)\s*[\xa0]\S*'
    
    # Регулярное выражение для поиска всех возможных сочетаний
    combinations_pattern = r'(?:рисунок\s+\d+|таблица\s+\d+|от\s+\d+\s+до\s+\d+|менее\s+\d+|свыше\s+\d+)'
    
    # Список для хранения найденных сочетаний
    found_combinations = []
    
    try:
        # Открываем PDF файл
        with open(pdf_path, 'rb') as file:
            pdf_reader = PyPDF2.PdfReader(file)
            
            # Проходим по каждой странице
            for page_num, page in enumerate(pdf_reader.pages, 1):
                # Извлекаем текст со страницы
                text = page.extract_text()
                
                # Разбиваем текст на строки
                lines = text.split('\n')
                
                # Проверяем каждую строку
                for line_num, line in enumerate(lines, 1):
                    # Ищем все совпадения специальных сочетаний
                    matches = re.finditer(combinations_pattern, line, re.IGNORECASE)
                    
                    for match in matches:
                        # Проверяем, есть ли неразрывный пробел (U+00A0) в найденном сочетании
                        matched_text = match.group()
                        
                        # Ищем неразрывный пробел в совпадении
                        if '\xa0' in matched_text:
                            found_combinations.append((matched_text, page_num, line_num))
    
    except Exception as e:
        print(f"Ошибка при обработке PDF-файла: {e}")
        return False, []
    
    # Если найдены сочетания с неразрывными пробелами - это означает нарушение
    if found_combinations:
        return True, found_combinations
    else:
        return False, []

def find_special_combinations_detailed(pdf_path: str) -> Tuple[bool, List[Tuple[str, int, int]]]:
    """
    Более точная версия функции для поиска специальных сочетаний.
    Ищет случаи, где неразрывный пробел НЕ используется в сочетаниях.
    """
    
    # Регулярное выражение для поиска специальных сочетаний
    # Ищем сочетания: "рисунок 1", "таблица 2", "от 1 до 5", "менее 5", "свыше 25"
    # Это выражение ищет сочетания, где между словом и числом нет неразрывного пробела
    # но при этом проверяет, что это именно те сочетания
    
    combinations_pattern = r'(?:рисунок\s+\d+|таблица\s+\d+|от\s+\d+\s+до\s+\d+|менее\s+\d+|свыше\s+\d+)'
    
    # Список для хранения найденных сочетаний
    found_combinations = []
    
    try:
        with open(pdf_path, 'rb') as file:
            pdf_reader = PyPDF2.PdfReader(file)
            
            # Проходим по каждой странице
            for page_num, page in enumerate(pdf_reader.pages, 1):
                # Извлекаем текст со страницы
                text = page.extract_text()
                
                # Разбиваем текст на строки
                lines = text.split('\n')
                
                # Проверяем каждую строку
                for line_num, line in enumerate(lines, 1):
                    # Ищем все совпадения специальных сочетаний
                    matches = re.finditer(combinations_pattern, line, re.IGNORECASE)
                    
                    for match in matches:
                        matched_text = match.group()
                        
                        # Проверяем, что в найденном сочетании НЕ используется неразрывный пробел
                        # В данном случае мы ищем "нарушения" - когда есть обычный пробел вместо неразрывного
                        # Но поскольку задача - проверить отсутствие неразрывного пробела,
                        # то "нарушение" будет означать, что неразрывный пробел отсутствует
                        # Это немного не соответствует формулировке задачи, поэтому лучше искать наоборот
                        
                        # Проверим, есть ли неразрывный пробел в сочетании
                        if '\xa0' in matched_text:
                            # Найден неразрывный пробел - это правильно
                            continue
                        else:
                            # Нет неразрывного пробела - это может быть проблемой
                            found_combinations.append((matched_text, page_num, line_num))
    
    except Exception as e:
        print(f"Ошибка при обработке PDF-файла: {e}")
        return False, []
    
    # Возвращаем результат
    if found_combinations:
        return True, found_combinations
    else:
        return False, []

def find_special_combinations_correct(pdf_path: str) -> Tuple[bool, List[Tuple[str, int, int]]]:
    """
    Правильная реализация для проверки отсутствия неразрывного пробела.
    Ищет случаи, где сочетания содержат обычный пробел вместо неразрывного.
    """
    
    # Регулярное выражение для поиска специальных сочетаний с обычным пробелом
    # Возвращает кортеж (наличие нарушения, список найденных сочетаний)
    combinations_pattern = r'(?:рисунок\s+\d+|таблица\s+\d+|от\s+\d+\s+до\s+\d+|менее\s+\d+|свыше\s+\d+)'
    
    # Список для хранения найденных сочетаний
    found_combinations = []
    
    try:
        with open(pdf_path, 'rb') as file:
            pdf_reader = PyPDF2.PdfReader(file)
            
            # Проходим по каждой странице
            for page_num, page in enumerate(pdf_reader.pages, 1):
                # Извлекаем текст со страницы
                text = page.extract_text()
                
                # Разбиваем текст на строки
                lines = text.split('\n')
                
                # Проверяем каждую строку
                for line_num, line in enumerate(lines, 1):
                    # Ищем все совпадения специальных сочетаний
                    matches = re.finditer(combinations_pattern, line, re.IGNORECASE)
                    
                    for match in matches:
                        matched_text = match.group()
                        
                        # Проверяем, что в найденном сочетании нет неразрывного пробела
                        # Если есть обычный пробел (который может быть разорван при переносе),
                        # это и есть проблема, которую нужно обнаружить
                        
                        # Для точной проверки нужно проверять, что сочетание не содержит \xa0
                        if '\xa0' not in matched_text:
                            # Нет неразрывного пробела - это проблема для переноса
                            found_combinations.append((matched_text, page_num, line_num))
    
    except Exception as e:
        print(f"Ошибка при обработке PDF-файла: {e}")
        return False, []
    
    # Возвращаем результат
    if found_combinations:
        return True, found_combinations
    else:
        return False, []

def main():
```
